// Factorial recursivo
fun factorial(n) {
    if (n <= 1) {
        return 1;
    }
    return n * factorial(n - 1);
}

// Factorial iterativo
fun factorialIterativo(n) {
    var result = 1;
    var i = 1;
    
    while (i <= n) {
        result = result * i;
        i = i + 1;
    }
    
    return result;
}

// Pruebas
print("Factorial recursivo de 5:");
print(factorial(5));

print("Factorial iterativo de 5:");
print(factorialIterativo(5));

print("Tabla de factoriales:");
var num = 0;
while (num <= 10) {
    print(num + "! = " + factorialIterativo(num));
    num = num + 1;
}
